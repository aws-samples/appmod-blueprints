  # ── CARM wiring (managed ACK IAMRoleSelector) — applied FIRST ──
  # The managed ACK capability routes a resource to a workload role via a
  # cluster-scoped IAMRoleSelector matching the CR's namespace; with no match it
  # falls back to the capability role (which only has AssumeWorkloadRoles +
  # ManageIRSARoles → cannot create SNS/SQS/AMP or a cluster-mgmt-* role).
  # These selectors MUST exist before any incident-bridge ACK CR's first reconcile:
  # the managed capability caches "no role selected" and does NOT re-evaluate a
  # selector added later (it would need a delete/recreate). incident-bridge-aws
  # therefore dependsOn this component. Once the selector is matched, a missing
  # target role is a normal Recoverable error that ACK retries — so role-creation
  # ordering inside incident-bridge-aws self-heals.
  - name: incident-bridge-selectors
    type: k8s-objects
    properties:
      objects:
      # iam CRs (both roles in incident-bridge-aws) -> cluster-mgmt-iam (IAMFullAccess),
      # which the capability role is already trusted to assume. Lets ACK create the
      # roles declaratively — no direct `aws iam` call.
      - apiVersion: services.k8s.aws/v1alpha1
        kind: IAMRoleSelector
        metadata:
          name: incident-bridge-iam
        spec:
          arn: arn:aws:iam::REPLACE_ACCOUNT_ID:role/REPLACE_CLUSTER_PREFIX-cluster-mgmt-iam
          namespaceSelector:
            names: [aiops-agent]
          resourceTypeSelector:
          - group: iam.services.k8s.aws
            kind: ""
            version: ""
      # sns/sqs/aps CRs -> the dedicated provisioning role (created below). It is
      # named <prefix>-cluster-mgmt-* so the capability's AssumeWorkloadRoles
      # (Resource: <prefix>-cluster-mgmt-*) permits sts:AssumeRole on it.
      - apiVersion: services.k8s.aws/v1alpha1
        kind: IAMRoleSelector
        metadata:
          name: incident-bridge-aws
        spec:
          arn: arn:aws:iam::REPLACE_ACCOUNT_ID:role/REPLACE_CLUSTER_PREFIX-cluster-mgmt-incident-bridge
          namespaceSelector:
            names: [aiops-agent]
          resourceTypeSelector:
          - group: sns.services.k8s.aws
            kind: ""
            version: ""
          - group: sqs.services.k8s.aws
            kind: ""
            version: ""
          - group: prometheusservice.services.k8s.aws
            kind: ""
            version: ""
      # eks CRs (the consumer's PodIdentityAssociation, and the agent-access
      # AccessEntries) -> cluster-mgmt-eks (has eks:*). The agent's OWN Pod Identity
      # comes from the aws-service-identity trait (Crossplane), so there was no prior

      - apiVersion: services.k8s.aws/v1alpha1

        metadata:
          name: incident-bridge-eks
        spec:
          arn: arn:aws:iam::REPLACE_ACCOUNT_ID:role/REPLACE_CLUSTER_PREFIX-cluster-mgmt-eks
          namespaceSelector:
            names: [aiops-agent]
          resourceTypeSelector:
          - group: eks.services.k8s.aws
            kind: ""
            version: ""
